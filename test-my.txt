DECLARE target_url STRING DEFAULT 'https://whichvoip.co.za/compare-voip-providers';

DECLARE data_start_date DATE DEFAULT (
  SELECT MIN(data_date) 
  FROM `telecoms-media-g-1754682898815.searchconsole.searchdata_url_impression`
  WHERE url LIKE CONCAT('%', target_url, '%')
);

DECLARE days_available INT64 DEFAULT DATE_DIFF(CURRENT_DATE(), data_start_date, DAY);

DECLARE current_period_start DATE DEFAULT (
  CASE 
    WHEN days_available >= 60 THEN DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
    ELSE DATE_SUB(CURRENT_DATE(), INTERVAL GREATEST(1, CAST(days_available / 4 AS INT64)) DAY)
  END
);

DECLARE previous_period_start DATE DEFAULT (
  CASE 
    WHEN days_available >= 60 THEN DATE_SUB(current_period_start, INTERVAL 30 DAY)
    ELSE DATE_SUB(current_period_start, INTERVAL GREATEST(1, CAST(days_available / 4 AS INT64)) DAY)
  END
);

WITH 
current_period AS (
  SELECT
    query,
    SUM(clicks) AS current_clicks,
    SUM(impressions) AS current_impressions,
    SAFE_DIVIDE(SUM(clicks), SUM(impressions)) AS current_ctr,
    SAFE_DIVIDE(SUM(sum_position), SUM(impressions)) AS current_avg_position
  FROM `telecoms-media-g-1754682898815.searchconsole.searchdata_url_impression`
  WHERE 
    url LIKE CONCAT('%', target_url, '%')
    AND data_date >= current_period_start
    AND data_date < CURRENT_DATE()
    AND is_anonymized_query = False
  GROUP BY query
),

previous_period AS (
  SELECT
    query,
    SUM(clicks) AS previous_clicks,
    SUM(impressions) AS previous_impressions,
    SAFE_DIVIDE(SUM(clicks), SUM(impressions)) AS previous_ctr,
    SAFE_DIVIDE(SUM(sum_position), SUM(impressions)) AS previous_avg_position
  FROM `telecoms-media-g-1754682898815.searchconsole.searchdata_url_impression`
  WHERE 
    url LIKE CONCAT('%', target_url, '%')
    AND data_date >= previous_period_start
    AND data_date < current_period_start
    AND is_anonymized_query = False
  GROUP BY query
),

combined_data AS (
  SELECT
    COALESCE(c.query, p.query) AS query,
    
    IFNULL(c.current_clicks, 0) AS current_clicks,
    IFNULL(p.previous_clicks, 0) AS previous_clicks,
    
    IFNULL(c.current_impressions, 0) AS current_impressions,
    IFNULL(p.previous_impressions, 0) AS previous_impressions,
    
    IFNULL(c.current_ctr, 0) AS current_ctr,
    IFNULL(p.previous_ctr, 0) AS previous_ctr,
    
    IFNULL(c.current_avg_position, 0) AS current_avg_position,
    IFNULL(p.previous_avg_position, 0) AS previous_avg_position,
    
    IFNULL(c.current_clicks, 0) - IFNULL(p.previous_clicks, 0) AS clicks_diff,
    IFNULL(c.current_impressions, 0) - IFNULL(p.previous_impressions, 0) AS impressions_diff,
    IFNULL(c.current_ctr, 0) - IFNULL(p.previous_ctr, 0) AS ctr_diff,
    IFNULL(p.previous_avg_position, 0) - IFNULL(c.current_avg_position, 0) AS position_diff,
    
    CASE 
      WHEN IFNULL(p.previous_clicks, 0) = 0 AND IFNULL(c.current_clicks, 0) > 0 THEN 100
      WHEN IFNULL(p.previous_clicks, 0) = 0 THEN 0
      ELSE ROUND((IFNULL(c.current_clicks, 0) - IFNULL(p.previous_clicks, 0)) / IFNULL(p.previous_clicks, 0) * 100, 2)
    END AS clicks_diff_pct,
    
    CASE 
      WHEN IFNULL(p.previous_impressions, 0) = 0 AND IFNULL(c.current_impressions, 0) > 0 THEN 100
      WHEN IFNULL(p.previous_impressions, 0) = 0 THEN 0
      ELSE ROUND((IFNULL(c.current_impressions, 0) - IFNULL(p.previous_impressions, 0)) / IFNULL(p.previous_impressions, 0) * 100, 2)
    END AS impressions_diff_pct,
    
    CASE 
      WHEN IFNULL(p.previous_ctr, 0) = 0 AND IFNULL(c.current_ctr, 0) > 0 THEN 100
      WHEN IFNULL(p.previous_ctr, 0) = 0 THEN 0
      ELSE ROUND((IFNULL(c.current_ctr, 0) - IFNULL(p.previous_ctr, 0)) / IFNULL(p.previous_ctr, 0) * 100, 2)
    END AS ctr_diff_pct
  FROM current_period c
  FULL OUTER JOIN previous_period p ON c.query = p.query
)

SELECT
  query,
  
  current_clicks,
  previous_clicks,
  clicks_diff,
  clicks_diff_pct,
  
  current_impressions,
  previous_impressions,
  impressions_diff,
  impressions_diff_pct,
  
  current_ctr,
  previous_ctr,
  ctr_diff,
  ctr_diff_pct,
  
  current_avg_position,
  previous_avg_position,
  position_diff,
  
  CASE
    WHEN clicks_diff > 0 THEN 'Gaining'
    WHEN clicks_diff < 0 THEN 'Declining'
    ELSE 'Stable'
  END AS trend_category

FROM combined_data
WHERE 
  (current_clicks > 0 OR previous_clicks > 0 OR current_impressions > 10 OR previous_impressions > 10)
ORDER BY 

  ABS(clicks_diff) DESC,
  current_impressions DESC;
